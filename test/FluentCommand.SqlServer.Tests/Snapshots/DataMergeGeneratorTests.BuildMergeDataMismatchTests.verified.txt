MERGE INTO [dbo].[member_user] AS t
USING
(
    VALUES
    ('00000000-0000-0000-0000-000000000000', N'test@email.com', N'Test User', N'Test', N'User'), 
    ('00000000-0000-0000-0000-000000000000', N'blah@email.com', N'Blah User', N'Blah', N'User'), 
    ('00000000-0000-0000-0000-000000000000', N'random@email.com', N'Random User', N'Random', N'User')
)
AS s
(
    [Id], [email_address], [display_name], [first_name], [last_name]
)
ON
(
    t.[email_address] = s.[email_address]
)
WHEN NOT MATCHED BY TARGET THEN 
    INSERT
    (
        [Id],
        [email_address],
        [display_name],
        [first_name],
        [last_name]
    )
    VALUES
    (
        s.[Id],
        s.[email_address],
        s.[display_name],
        s.[first_name],
        s.[last_name]
    )
WHEN MATCHED THEN 
    UPDATE SET
        t.[Id] = s.[Id],
        t.[display_name] = s.[display_name],
        t.[first_name] = s.[first_name],
        t.[last_name] = s.[last_name]
;
